% section headline with coloured underline
\newcommand{\cvsection}[1]{%
  \vspace{9pt}%
  {\raggedright\textbf{\large\textcolor{darkcol}{\MakeUppercase{#1}}}\par

% contact line: icon name, text
\newcommand{\cvcontact}[2]{%
  \noindent\makebox[1.5em][l]{\textcolor{maincol}{\csname fa#1\endcsname}}%
  \parbox[t]{\dimexpr\linewidth-1.5em\relax}{\raggedright #2}\par\vspace{5pt}}

% small label above a value (personal details)
\newcommand{\cvdetail}[2]{%
  {\footnotesize\bfseries\textcolor{maincol}{\MakeUppercase{#1}}}\par
  {\raggedright #2\par}\vspace{5pt}}

% skill with bar: name, level label, fill 0..1
\newcommand{\cvskill}[3]{%
  \noindent\textbf{#1}\hfill\textcolor{maincol}{\small #2}\par\vspace{2pt}%
  \noindent\textcolor{lightcol}{\rule{\linewidth}{3pt}}\hspace{-\linewidth}%
  \textcolor{maincol}{\rule{#3\linewidth}{3pt}}\par\vspace{5pt}}

% name on the left, value on the right (languages)
\newcommand{\cvpair}[2]{\noindent\textbf{#1}\hfill\textcolor{darkcol}{#2}\par\vspace{4pt}}

% job entry header: dates, title, employer

    \textcolor{maincol}{\textbf{#3}} & \\
  \end{tabular*}\par\vspace{4pt}}
